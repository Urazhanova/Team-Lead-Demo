### Шаг 8: Реализовать Scoring (1.5 часа)

**8.1 Добавьте validateSelection() метод:**

```javascript
validateSelection: function() {
  var result = {
    isValid: true,
    errors: []
  };

  if (this.gameState.selectedTasks.length < 2) {
    result.errors.push("Выберите минимум 2 задачи");
    result.isValid = false;
  }

  if (this.gameState.usedCapacity > this.gameState.totalCapacity) {
    result.errors.push("Превышена пропускная способность");
    result.isValid = false;
  }

  return result;
}
```

**8.2 Добавьте calculateScore() метод:**

```javascript
calculateScore: function() {
  var score = 0;
  var criteriaScore = 25;  // Каждый критерий = 25%

  // Критерий 1: Критические задачи
  if (this.hasCriticalTasks()) {
    score += criteriaScore;
  }

  // Критерий 2: Technical debt
  if (this.hasTechDebtTasks()) {
    score += criteriaScore;
  }

  // Критерий 3: Баланс
  if (this.hasGoodBalance()) {
    score += criteriaScore;
  }

  // Критерий 4: Capacity
  if (this.gameState.usedCapacity <= this.gameState.totalCapacity) {
    score += criteriaScore;
  }

  return Math.round(score);
}
```

completeGame: function() {
  console.log("[GameContent] Completing game");
  this.gameState.currentPhase = 'review';
  this.gameState.endTime = Date.now();

  // Валидировать
  var validation = this.validateSelection();
  if (!validation.isValid) {

### Шаг 10: Добавить SCORM логирование (30 мин)

**10.1 Добавьте logToSCORM() метод:**

```javascript
logToSCORM: function() {
  if (typeof SCORM === 'undefined' || !SCORM.initialized) {
    console.warn("[GameContent] SCORM not available");
    return;
  }

  // Log interaction
  SCORM.addInteraction(
    'sprint-planning-3',
    'choice',
    this.gameState.selectedTasks.join(','),
    this.gameState.score >= 75
  );

  // Set score
  SCORM.setValue('cmi.core.score.raw', this.gameState.score);

  // Commit
  SCORM.commit();
}
```
