```javascript
// В drawGame()
function drawZones() {
    Object.values(GameData.zones).forEach(zone => {
        // Рисуем границу зоны (пунктирная линия)
        ctx.strokeStyle = 'rgba(78, 204, 163, 0.3)';
        ctx.setLineDash([5, 5]);
        ctx.strokeRect(zone.x, zone.y, zone.width, zone.height);
        ctx.setLineDash([]);

        // Рисуем метку зоны
        ctx.fillStyle = 'rgba(255, 255, 255, 0.5)';
        ctx.font = 'bold 12px Arial';
        ctx.fillText(zone.label, zone.x + 10, zone.y + 20);
    });
}

// В updateGameState()
function checkZoneEntry() {
    Object.entries(GameData.zones).forEach(([zoneKey, zone]) => {
        const isInZone = gameState.player.x >= zone.x &&
                        gameState.player.x <= zone.x + zone.width &&
                        gameState.player.y >= zone.y &&
                        gameState.player.y <= zone.y + zone.height;

        if (isInZone && gameState.currentZone !== zoneKey) {
            onZoneEnter(zoneKey);
        }
    });
}
```
